NAME
    ls - list directory contents.

SYNOPSIS
    ls [options] [path]

OPTIONS
    -a, --all      include hidden files and dot entries.
    -l, --long     print file attributes, size and name.
    -h, --help     print command help.

EXAMPLES
    ls
    ls -l C:\src
    ls --all
