# Wsh companion utilities. They are intentionally small standalone tools so
# they can also be executed from cmd.exe/PowerShell during diagnostics.
add_executable(ls ls.cpp)
add_executable(md md.cpp)
add_executable(tree tree.cpp)
add_executable(ping ping.cpp)
add_executable(wshinit wshinit.cpp)
add_executable(cat coreutils.cpp)
add_executable(pwd coreutils.cpp)
add_executable(cp coreutils.cpp)
add_executable(mv coreutils.cpp)
add_executable(rm coreutils.cpp)
add_executable(mkdir coreutils.cpp)
add_executable(rmdir coreutils.cpp)
add_executable(touch coreutils.cpp)
add_executable(head coreutils.cpp)
add_executable(tail coreutils.cpp)
add_executable(wc coreutils.cpp)
add_executable(grep coreutils.cpp)
add_executable(sort coreutils.cpp)
add_executable(uniq coreutils.cpp)
add_executable(basename coreutils.cpp)
add_executable(dirname coreutils.cpp)
add_executable(whoami coreutils.cpp)
add_executable(hostname coreutils.cpp)
add_executable(uname coreutils.cpp)
add_executable(date coreutils.cpp)
add_executable(clear coreutils.cpp)
add_executable(which coreutils.cpp)
add_executable(sleep coreutils.cpp)
add_executable(yes coreutils.cpp)
add_executable(true coreutils.cpp)
add_executable(false coreutils.cpp)
add_executable(echo coreutils.cpp)
add_executable(env coreutils.cpp)
add_executable(printenv coreutils.cpp)
add_executable(more coreutils.cpp)
add_executable(less coreutils.cpp)
add_executable(find coreutils.cpp)
add_executable(tee coreutils.cpp)
add_executable(cut coreutils.cpp)

foreach(tool ls md tree ping wshinit cat pwd cp mv rm mkdir rmdir touch head tail wc grep sort uniq basename dirname whoami hostname uname date clear which sleep yes true false echo env printenv more less find tee cut)
    target_compile_definitions(${tool} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(${tool} PRIVATE ${CMAKE_SOURCE_DIR}/src/core)
    target_link_libraries(${tool} PRIVATE wsh_core shell32)
    set_target_properties(${tool} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/dist"
        RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/dist"
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/dist"
    )
endforeach()
